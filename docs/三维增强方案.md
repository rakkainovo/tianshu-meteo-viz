### 已实现功能
- ✅ 基于 Three.js 的三维沙盘
- ✅ 立体行政区地图（OrthographicCamera + 正轴投影）
- ✅ 预警数量柱（高度映射）+ 预警点标注
- ✅ 全国→省→市三级下钻
- ✅ 历史回放（10分钟一帧）
- ✅ WebGL 上下文丢失降级处理

### 技术栈
- `three.module.js` + `OrbitControls.js`
- 正轴投影：`(lon-105)*cos(34°), 35-lat`
- 材质：面用 MeshLambertMaterial，柱用 ExtrudeGeometry
- 光照：HemisphereLight + DirectionalLight

#### 实现方案
```javascript
// 1. 预处理：GeoTIFF → 降采样 → JSON 高程网格
// Python：rasterio + numpy
// 输出：data/terrain/{adcode}.json
// 格式：{ width, height, min, max, data: Float32Array }

// 2. 前端渲染
import { project } from './sand-scene.js';

function buildTerrainMesh(geo, elevationGrid) {
  const { width, height, min, max, data } = elevationGrid;
  const geometry = new THREE.PlaneGeometry(
    geo.bbox.width, geo.bbox.height, width - 1, height - 1
  );
  
  // 修改顶点 Z 坐标
  const positions = geometry.attributes.position;
  for (let i = 0; i < positions.count; i++) {
    const elevation = data[i];
    positions.setZ(i, (elevation - min) / (max - min) * 5); // 夸张系数 5
  }
  
  // 法线重算
  geometry.computeVertexNormals();
  
  // 纹理：卫星影像 or 高程色带
  const material = new THREE.MeshStandardMaterial({
    map: satelliteTexture,
    displacementMap: elevationTexture,
    displacementScale: 5
  });
  
  return new THREE.Mesh(geometry, material);
}
```

#### 2.2 风场粒子系统
```javascript
// 基于 WebGL 的粒子追踪
// 参考：earth.nullschool.net 的风场可视化

class WindParticleSystem {
  constructor(windData, count = 10000) {
    this.windData = windData; // { u, v, w } 速度分量
    this.particles = new Float32Array(count * 3); // xyz 位置
    
    // 粒子几何体
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', 
      new THREE.BufferAttribute(this.particles, 3)
    );
    
    // 着色器：速度 → 颜色
    const material = new THREE.ShaderMaterial({
      uniforms: { windTexture: { value: windData.texture } },
      vertexShader: WIND_VERTEX_SHADER,
      fragmentShader: WIND_FRAGMENT_SHADER
    });
    
    this.points = new THREE.Points(geometry, material);
  }
  
  update(dt) {
    for (let i = 0; i < this.particles.length; i += 3) {
      const [x, y, z] = [this.particles[i], this.particles[i+1], this.particles[i+2]];
      const [u, v, w] = this.sampleWind(x, y, z);
      
      // 欧拉积分
      this.particles[i] += u * dt;
      this.particles[i+1] += v * dt;
      this.particles[i+2] += w * dt;
      
      // 边界重置
      if (outOfBounds(x, y, z)) {
        this.resetParticle(i / 3);
      }
    }
    
    this.points.geometry.attributes.position.needsUpdate = true;
  }
}
```

### 方向 3：时空立方体（Space-Time Cube）

#### 概念
将 **空间（x, y）+ 时间（t）** 映射为三维坐标：
- XY 平面 = 地理位置
- Z 轴 = 时间（过去在底部，现在在顶部）

#### 实现
```javascript
function buildSpaceTimeCube(alarms, timeWindow) {
  const group = new THREE.Group();
  
  alarms.forEach(alarm => {
    const [x, y] = project(alarm.lng, alarm.lat);
    const z = (alarm.publishTime - timeWindow.start) / timeWindow.duration * 10;
    
    // 预警事件用柱体表示（从地面长到事件时刻）
    const geometry = new THREE.CylinderGeometry(0.1, 0.1, z, 8);
    const material = new THREE.MeshLambertMaterial({
      color: getLevelColor(alarm.level),
      transparent: true,
      opacity: 0.7
    });
    
    const mesh = new THREE.Mesh(geometry, material);
    mesh.position.set(x, z / 2, -y);
    group.add(mesh);
  });
  
  return group;
}
```

#### 效果
- 从侧面看：时间演变的"瀑布"
- 从顶部看：当前态势分布
- 聚类分析：预警时空热点一目了然

```javascript
// 基于台风强度生成程式化云墙模型
function buildTyphoonCloud(typhoon) {
  const radius = typhoon.radius7 / 111; // 7 级风圈半径（度）
  const height = 12 + typhoon.strong * 0.5; // 强度越大云顶越高
  
  // 使用 Noise 纹理生成云层效果
  const geometry = new THREE.CylinderGeometry(
    radius * 0.3,  // 风眼
    radius,        // 底部
    height,
    32, 16
  );
  
  const material = new THREE.MeshPhongMaterial({
    map: cloudTexture,
    transparent: true,
    opacity: 0.6,
    side: THREE.DoubleSide
  });
  
  // 顶点扰动（模拟云层不规则性）
  const positions = geometry.attributes.position;
  for (let i = 0; i < positions.count; i++) {
    const noise = simplex3(
      positions.getX(i) * 0.1,
      positions.getY(i) * 0.1,
      positions.getZ(i) * 0.1
    );
    positions.setY(i, positions.getY(i) + noise * 2);
  }
  
  geometry.computeVertexNormals();
  return new THREE.Mesh(geometry, material);
}
```

**4.2 台风路径管道**
```javascript
// 用 TubeGeometry 替代平面线段
function buildTyphoonTube(path) {
  const points = path.map(p => new THREE.Vector3(...project(p.lng, p.lat), p.strong * 0.5));
  const curve = new THREE.CatmullRomCurve3(points);
  
  const geometry = new THREE.TubeGeometry(
    curve,
    64,     // 分段数
    0.15,   // 半径
    8,      // 径向分段
    false
  );
  
  // 颜色渐变：起点→终点 = 蓝→红
  const colors = new Float32Array(geometry.attributes.position.count * 3);
  for (let i = 0; i < colors.length; i += 3) {
    const t = (i / 3) / (colors.length / 3);
    colors[i] = t;       // R
    colors[i+1] = 0;     // G
    colors[i+2] = 1 - t; // B
  }
  geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));
  
  const material = new THREE.MeshBasicMaterial({ vertexColors: true });
  return new THREE.Mesh(geometry, material);
}
```

def extract_region_elevation(tif_path, bbox, target_size=128):
    """从 GeoTIFF 提取区域高程并降采样"""
    with rasterio.open(tif_path) as src:
        # 裁剪边界框
        window = rasterio.windows.from_bounds(*bbox, src.transform)
        elevation = src.read(1, window=window)
        
        # 降采样到固定尺寸
        from scipy.ndimage import zoom
        scale = min(target_size / elevation.shape[0], target_size / elevation.shape[1])

        return {
            "width": elevation_small.shape[1],
            "height": elevation_small.shape[0],
            "min": float(elevation_small.min()),
            "max": float(elevation_small.max()),
            "data": elevation_small.flatten().tolist()
        }

## 五、性能优化建议

### 当前性能瓶颈
- 行政区划 Mesh 顶点数过多（海岸线精度高）
- 大量预警点的粒子系统卡顿

### 优化方案

**1. LOD（Level of Detail）**
```javascript
// 远景用低精度模型，近景用高精度
import { LOD } from 'three';

const lod = new LOD();
lod.addLevel(highDetailMesh, 0);    // 0-5 单位距离
lod.addLevel(mediumDetailMesh, 5);  // 5-15
lod.addLevel(lowDetailMesh, 15);    // 15+

scene.add(lod);
```

**2. GPU Instancing**
```javascript
// 大量相同几何体（预警点）用 InstancedMesh
const geometry = new THREE.SphereGeometry(0.1);
const material = new THREE.MeshBasicMaterial();
const count = alarms.length;

const mesh = new THREE.InstancedMesh(geometry, material, count);

alarms.forEach((alarm, i) => {
  const matrix = new THREE.Matrix4();
  matrix.setPosition(...project(alarm.lng, alarm.lat), 0.5);
  mesh.setMatrixAt(i, matrix);
  mesh.setColorAt(i, new THREE.Color(getLevelColor(alarm.level)));
});

scene.add(mesh);
```

**3. 八叉树空间索引**

## 七、技术债务与坑点

### 已知问题
1. **坐标投影简化** - 当前用简单的正轴投影，边界省份（新疆、黑龙江）变形明显
   - 解决：改用 d3-geo 的 Albers 投影

2. **WebGL 上下文限制** - 移动设备可能丢失上下文
   - 已有降级方案，但体验需优化

3. **内存泄漏风险** - 频繁切换地图不释放 Geometry/Material
   - 需要完善 `disposeTree` 调用时机

### 数据获取难点
- **实时风场数据**：气象局不公开，需要付费 API 或模拟数据
- **雷达数据**：SWAN 格式解析复杂，且数据量巨大（单个体扫 ~10MB）
